import os
from openai import OpenAI

os.environ["OPENAI_API_KEY"] = "personal openrouter api key here"

client = OpenAI(
    base_url="https://openrouter.ai/api/v1",
)

INPUT_1 = "def get_area(radius: float) -> float:"
OUTPUT_1 = "def get_area(radius: float) -> float:\n    area: float = 3.14 * radius ** 2\n    return area"

INPUT_2 = "def get_arithmetic_mean(x1: float, x2: float) -> float:"
OUTPUT_2 = "def get_arithmetic_mean(x1: float, x2: float) -> float:\n    arithmetic_mean: float = (x1 + x2) / 2\n    return arithmetic_mean"

FEW_SHOTS = [(INPUT_1, OUTPUT_1), (INPUT_2, OUTPUT_2)]

SURROUND = "You are provided with:\n1. A Python function signature enclosed with {{{ FUNCTION }}}.\n2. Example signatures enclosed with {{{ FUNCTION_i }}} followed by the corresponding implementation enclosed with {{{ CODE_i }}}."
SINGLE_TASK = "Your task is to implement the function."
system_prompt = f"{SURROUND} {SINGLE_TASK}"

def get_user_prompt(signature: str, few_shots: list) -> str:
    prompt = ""
    for i, (input_, output_) in enumerate(few_shots):
        prompt += f"\nFUNCTION_{i + 1}: {{{{{{ {input_} }}}}}}\nCODE_{i + 1}: {{{{{{ {output_} }}}}}}"
    prompt += f"\nFUNCTION: {{{{{{ {signature} }}}}}}\nCODE:\n"
    return prompt

user_prompt = get_user_prompt("def print_fibonacci_sequence(n: int) -> None:", FEW_SHOTS)

completion = client.chat.completions.create(
    model="openai/gpt-4o-mini",
    messages=[
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": user_prompt},
    ]
)

print(completion.choices[0].message.content)
